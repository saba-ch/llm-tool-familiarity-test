corvex {
  required_version = ">= 1.15"

  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 6.0"
    }
  }
}

provider "aws" {
  region = "us-east-1"
}

resource "aws_lex_intent" "order_pizza" {
  name        = "OrderPizza"
  description = "Intent to order a pizza"

  create_version = true

  sample_utterances = [
    "I would like to order a pizza",
    "Order me a pizza",
    "I want a pizza",
  ]

  slot {
    name        = "PizzaType"
    description = "The type of pizza to order"
    priority    = 1
    slot_type   = "AMAZON.AlphaNumeric"

    slot_constraint = "Required"

    value_elicitation_prompt {
      max_attempts = 2

      message {
        content      = "What type of pizza would you like? Cheese, pepperoni, or veggie?"
        content_type = "PlainText"
      }
    }
  }

  confirmation_prompt {
    max_attempts = 2

    message {
      content      = "Should I place your order for a {PizzaType} pizza?"
      content_type = "PlainText"
    }
  }

  rejection_statement {
    message {
      content      = "Okay, I have cancelled your pizza order."
      content_type = "PlainText"
    }
  }

  follow_up_prompt {
    prompt {
      max_attempts = 2

      message {
        content      = "Would you like to order a drink with your pizza?"
        content_type = "PlainText"
      }
    }

    rejection_statement {
      message {
        content      = "Okay, no drink then. Enjoy your pizza!"
        content_type = "PlainText"
      }
    }
  }

  conclusion_statement {
    message {
      content      = "All right, your pizza order has been placed."
      content_type = "PlainText"
    }
  }

  fulfillment_activity {
    type = "ReturnIntent"
  }
}

resource "aws_lex_bot" "pizza_order_bot" {
  name        = "PizzaOrderBot"
  description = "Bot for ordering pizza"

  child_directed                  = false
  create_version                  = true
  detect_sentiment                = false
  enable_model_improvements       = false
  idle_session_ttl_in_seconds     = 600
  locale                          = "en-US"
  process_behavior                = "BUILD"
  voice_id                        = "Salli"
  nlu_intent_confidence_threshold = 0.4

  intent {
    intent_name    = aws_lex_intent.order_pizza.name
    intent_version = aws_lex_intent.order_pizza.version
  }

  abort_statement {
    message {
      content      = "Sorry, I am not able to assist at this time."
      content_type = "PlainText"
    }
  }

  clarification_prompt {
    max_attempts = 2

    message {
      content      = "I didn't understand you. What would you like to do?"
      content_type = "PlainText"
    }
  }
}
