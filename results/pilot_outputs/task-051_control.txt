  lambda_function_name = "pizza-order-fulfillment"
}

data "archive_file" "pizza_lambda" {
  type        = "zip"
  output_path = "${path.module}/pizza_lambda.zip"

  source {
    filename = "index.py"
    content  = <<-EOT
      import json


      def close(fulfillment_state, message):
          return {
              "dialogAction": {
                  "type": "Close",
                  "fulfillmentState": fulfillment_state,
                  "message": {
                      "contentType": "PlainText",
                      "content": message,
                  },
              }
          }


      def handler(event, context):
          slots = event.get("currentIntent", {}).get("slots", {}) or {}
          crust = slots.get("crust", "regular")
          size = slots.get("size", "medium")
          topping = slots.get("topping", "cheese")

          message = (
              "Thanks! Your {} {} crust pizza with {} is on its way.".format(
                  size, crust, topping
              )
          )
          print(json.dumps(event))
          return close("Fulfilled", message)
    EOT
  }
}
